@OptIn(ExperimentalMaterial3Api::class)

fun BroMainScreen() {

        mutableStateListOf(

    Scaffold(

        bottomBar = {
            Surface(
                color = Color(0xFF1E1E1E),
                modifier = Modifier.fillMaxWidth()
            ) {
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(8.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    IconButton(onClick = { /* TODO: Voice Input in Stage 3 */ }) {
                        Icon(Icons.Default.Mic, contentDescription = "Voice Input", tint = Color.Cyan)
                    }
                    OutlinedTextField(
                        value = textInput,
                        onValueChange = { textInput = it },
                        modifier = Modifier
                            .weight(1f)
                            .padding(horizontal = 4.dp),
                        placeholder = { Text("Ask BRO...", color = Color.Gray) },
                        colors = TextFieldDefaults.outlinedTextFieldColors(
                            focusedBorderColor = Color.Cyan,
                            unfocusedBorderColor = Color.DarkGray,
                            cursorColor = Color.Cyan
                        ),
                        maxLines = 1
                    )
                    IconButton(onClick = {
                        if (textInput.isNotBlank()) {
                            chatMessages.add(ChatMessage(textInput, true))
                            val query = textInput
                            textInput = ""
                            // Simulate response for Stage 1
                            chatMessages.add(ChatMessage("Received: $query", false))
                        }
                    }) {
                        Icon(Icons.Default.Send, contentDescription = "Send", tint = Color.Cyan)
                    }
                }
            }
        }

        Column(

                .fillMaxSize()
                .background(Color(0xFF121212))
                .padding(innerPadding)

                    .fillMaxWidth()

                Text("🔵 [BRO Orb Animation State: IDLE]", color = Color.Cyan)

            // Chat Messages List
            LazyColumn(
                modifier = Modifier

                    .weight(1f)
                    .padding(8.dp),
                verticalArrangement = Arrangement.spacedBy(8.dp)

                    Box(

                        Surface(

                            shape = MaterialTheme.shapes.medium

                            Text(
                                text = message.text,
                                color = Color.White,
                                modifier = Modifier.padding(12.dp)
                            )
                        }
                    }
                }
            }
        }
    }

    onSurface = Color.White,
